 def test_handoff_guard_accepts_bound_receipt_and_rejects_missing_or_changed(self):

  with tempfile.TemporaryDirectory() as tmp:
   workspace=Path(tmp)/'.coding-agent/tickets/FIX';workspace.mkdir(parents=True);path=workspace/'state.json'
   old={'ticket_id':'FIX','ticket_type':'bugfix','current_state':'ANALYSIS','states':{'ANALYSIS':{'status':'in_progress'}}};path.write_text(json.dumps(old))
   artifact=workspace/'suspect-labels.json';candidate=copy.deepcopy(old);candidate['current_state']='PLANNING';candidate['states']['ANALYSIS']['status']='completed'
   def invoke():
    event={'tool_name':'Write','cwd':tmp,'tool_input':{'file_path':str(path),'content':json.dumps(candidate)}}
    return subprocess.run([sys.executable,str(ROOT/'hooks/suspect-labeling-guard.py')],input=json.dumps(event),text=True,capture_output=True)
   self.assertIn('deny',invoke().stdout)
   artifact.write_text(json.dumps(labels.append_round(None,response(),'no_new_candidates')))
   self.assertIn('deny',invoke().stdout)
   candidate['states']['ANALYSIS']['suspect_labels_sha256']=hashlib.sha256(artifact.read_bytes()).hexdigest()
   self.assertEqual(invoke().stdout,'')
   artifact.write_text(artifact.read_text()+' ');self.assertIn('deny',invoke().stdout)
   candidate['ticket_type']='fresh';self.assertIn('deny',invoke().stdout)
