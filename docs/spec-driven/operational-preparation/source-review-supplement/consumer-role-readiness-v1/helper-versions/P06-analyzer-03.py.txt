def proxy(args):

        with lock,args.log.open('a') as f:f.write(json.dumps({'kind':kind,'time':time.time(),**data})+'\n')
    command=[str(args.binary),'mcp','--config',str(args.config)]
    child=subprocess.Popen(command,stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
    record('start',wrapper_pid=os.getpid(),child_pid=child.pid,command=command,binary_sha256=sha(args.binary))
    def eof():
        with input_lock:
            if not child.stdin.closed:
                try:child.stdin.close()
                except OSError:pass
                record('stdin_eof')
    def stop(number,_frame):record('signal',signal=number);eof()
    for number in [signal.SIGTERM,signal.SIGINT]:signal.signal(number,stop)
    def read_input():
        count=0
        try:
            for raw in sys.stdin.buffer:
                row=json.loads(raw)
                if row.get('method')=='tools/call':
                    count+=1
                    if count>64:record('budget_refused',tool_requests=count);break
                record('request',raw=raw.decode())

        except (OSError,ValueError) as e:record('input_end',reason=type(e).__name__)

    def read_output():

                record('response',raw=raw.decode());sys.stdout.buffer.write(raw);sys.stdout.buffer.flush()
        except (BrokenPipeError,OSError):record('output_closed');eof()
    def read_error():
        for raw in child.stderr:record('stderr',raw=raw.decode())
    threads=[threading.Thread(target=f,daemon=True) for f in [read_input,read_output,read_error]]
    for t in threads:t.start()
