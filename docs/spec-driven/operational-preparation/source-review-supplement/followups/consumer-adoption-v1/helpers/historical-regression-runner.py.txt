import concurrent.futures,json,os,re,subprocess,time
from pathlib import Path
root=Path('/private/tmp/ks-adoption-historical-context-20261017')
tests=json.load(open(root/'docs/spec-driven/OPERATIONAL-ACCEPTANCE-PROGRESS.json'))['phases'][0]['preparation_validation']['consumer_final']['regression_tests']
folder=Path('/private/tmp/ks-adoption-historical-regressions');folder.mkdir(exist_ok=False)
def run(item):
 name,count=item
 start=time.monotonic()
 with (folder/(name+'.log')).open('wb') as f:
  p=subprocess.run(['python3',str(root/'scripts'/name)],cwd=root,stdout=f,stderr=subprocess.STDOUT,env={**os.environ,'PYTHONDONTWRITEBYTECODE':'1','GOCACHE':'/private/tmp/ks-adoption-go-cache-20261017'})
 text=(folder/(name+'.log')).read_text()
 actual=re.search(r'Ran (\d+) tests?',text)
 return {'name':name,'expected':count,'observed':int(actual.group(1)) if actual else None,'exit_code':p.returncode,'ok':p.returncode==0 and actual is not None and int(actual.group(1))==count and '\nOK' in text,'elapsed_seconds':time.monotonic()-start}
with concurrent.futures.ThreadPoolExecutor(max_workers=3)as pool:
 results=list(pool.map(run,tests.items()))
(folder/'results.json').write_text(json.dumps(results,indent=2)+'\n')
print(json.dumps({'suites':len(results),'tests':sum(r['expected']for r in results),'pass':all(r['ok']for r in results),'failed':[r['name']for r in results if not r['ok']]}))
