// Package experiment freezes timing/fault/observation investigations before
// execution. It distinguishes the designated symptom from infrastructure errors.
package experiment

import (
	"crypto/sha256"
	"encoding/binary"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"strings"
)

type Source struct {
	Commit            string `json:"commit"`
	BinarySHA256      string `json:"binary_sha256"`
	DriverSHA256      string `json:"driver_sha256"`
	EnvironmentSHA256 string `json:"environment_sha256"`
}

type Injection struct {
	Kind       string `json:"kind"`   // delay, restart, partition; observation never injects
	Target     string `json:"target"` // node id inside the trial-owned network
	Step       string `json:"step"`
	MinDelayMS int    `json:"min_delay_ms"`
	MaxDelayMS int    `json:"max_delay_ms"`
}

type ResumeTrigger struct {
	Event         string `json:"event"`
	AssertionID   string `json:"assertion_id"`
	WindowSeconds int    `json:"window_seconds"`
}

type Plan struct {
	SchemaVersion       int            `json:"schema_version"`
	ID                  string         `json:"id"`
	Mode                string         `json:"mode"` // reproduction, comparison, observation
	Symptom             string         `json:"symptom"`
	AssertionID         string         `json:"assertion_id"`
	Seed                string         `json:"seed"`
	Before              Source         `json:"before"`
	After               *Source        `json:"after"`
	Repetitions         int            `json:"repetitions"`
	MinCompleted        int            `json:"min_completed"`
	MinSymptomFailures  int            `json:"min_symptom_failures"`
	MaxFailureRate      float64        `json:"max_failure_rate"`
	TrialTimeoutSeconds int            `json:"trial_timeout_seconds"`
	TotalTimeoutSeconds int            `json:"total_timeout_seconds"`
	OwnedNodeIDs        []string       `json:"owned_node_ids"`
	NormalAssertionIDs  []string       `json:"normal_assertion_ids"`
	Injection           *Injection     `json:"injection"`
	Resume              *ResumeTrigger `json:"resume"`
}

type Schedule struct {
	Trial   int    `json:"trial"`
	Seed    string `json:"seed"`
	DelayMS int    `json:"delay_ms"`
}

func validHash(s string, n int) bool {
	if len(s) != n || strings.ToLower(s) != s {
		return false
	}
	_, e := hex.DecodeString(s)
	return e == nil
}
func (s Source) validate() error {
	if !validHash(s.Commit, 40) || !validHash(s.BinarySHA256, 64) || !validHash(s.DriverSHA256, 64) || !validHash(s.EnvironmentSHA256, 64) {
		return fmt.Errorf("source, binary, driver and environment pins required")
	}
	return nil
}
func unique(values []string) bool {
	seen := map[string]bool{}
	for _, v := range values {
		if v == "" || len(v) > 128 || seen[v] {
			return false
		}
		seen[v] = true
	}
	return true
}
func contains(values []string, value string) bool {
	for _, v := range values {
		if v == value {
			return true
		}
	}
	return false
}
func (p Plan) Validate() error {
	if p.SchemaVersion != 1 || p.ID == "" || len(p.ID) > 128 || strings.TrimSpace(p.Symptom) == "" || len(p.Symptom) > 4096 || p.AssertionID == "" || len(p.AssertionID) > 128 || p.Seed == "" || len(p.Seed) > 128 {
		return fmt.Errorf("invalid experiment identity")
	}
	if p.Mode != "reproduction" && p.Mode != "comparison" && p.Mode != "observation" {
		return fmt.Errorf("unsupported experiment mode")
	}
	if e := p.Before.validate(); e != nil {
		return e
	}
	if (p.Mode == "comparison") != (p.After != nil) {
		return fmt.Errorf("comparison requires distinct frozen before and after sources")
	}
	if p.After != nil {
		if e := p.After.validate(); e != nil {
			return e
		}
		if p.After.Commit == p.Before.Commit || p.After.DriverSHA256 != p.Before.DriverSHA256 || p.After.EnvironmentSHA256 != p.Before.EnvironmentSHA256 {
			return fmt.Errorf("comparison sources must differ under the same driver/environment")
		}
	}
	if p.Repetitions < 1 || p.Repetitions > 1000 || p.MinCompleted < 1 || p.MinCompleted > p.Repetitions || p.MinSymptomFailures < 1 || p.MinSymptomFailures > p.Repetitions || !(p.MaxFailureRate > 0 && p.MaxFailureRate < 1) {
		return fmt.Errorf("invalid frozen denominator/rate threshold")
	}
	if p.TrialTimeoutSeconds < 1 || p.TrialTimeoutSeconds > 300 || p.TotalTimeoutSeconds < 1 || p.TotalTimeoutSeconds > 7200 {
		return fmt.Errorf("invalid time budget")
	}
	if len(p.NormalAssertionIDs) < 1 || len(p.NormalAssertionIDs) > 16 || !unique(p.NormalAssertionIDs) || contains(p.NormalAssertionIDs, p.AssertionID) {
		return fmt.Errorf("distinct normal control assertions required")
	}
	if len(p.OwnedNodeIDs) > 32 || !unique(p.OwnedNodeIDs) {
		return fmt.Errorf("invalid owned node set")
	}
	if p.Mode == "observation" {
		if p.Injection != nil || p.Resume == nil || (p.Resume.Event == "" || len(p.Resume.Event) > 128) || p.Resume.AssertionID != p.AssertionID || p.Resume.WindowSeconds < 1 || p.Resume.WindowSeconds > 3600 {
			return fmt.Errorf("observation requires bounded designated resume trigger and no injection")
		}
		if p.Resume.WindowSeconds > p.TotalTimeoutSeconds || (p.Resume.WindowSeconds+p.Repetitions-1)/p.Repetitions >= p.TrialTimeoutSeconds {
			return fmt.Errorf("observation window must fit total and per-trial budgets with execution margin")
		}
	} else {
		if p.Resume != nil || p.Injection == nil {
			return fmt.Errorf("reproduction/comparison needs explicit injection")
		}
		i := p.Injection
		if (i.Kind != "delay" && i.Kind != "restart" && i.Kind != "partition") || !contains(p.OwnedNodeIDs, i.Target) || i.Step == "" || len(i.Step) > 128 || i.MinDelayMS < 0 || i.MaxDelayMS < i.MinDelayMS || i.MaxDelayMS > 60000 {
			return fmt.Errorf("injection must target a trial-owned node at a bounded named step")
		}
	}
	return nil
}

// Schedules are derived by a specified hash algorithm, not a language's evolving
// RNG implementation. Before and after use exactly the same trial schedules.
func (p Plan) Schedules() ([]Schedule, error) {
	if e := p.Validate(); e != nil {
		return nil, e
	}
	schedules := make([]Schedule, p.Repetitions)
	for i := range schedules {
		sum := sha256.Sum256([]byte(fmt.Sprintf("chainbench-experiment-v1\x00%s\x00%d", p.Seed, i+1)))
		delay := 0
		if p.Injection != nil {
			width := uint64(p.Injection.MaxDelayMS - p.Injection.MinDelayMS + 1)
			delay = p.Injection.MinDelayMS + int(binary.BigEndian.Uint64(sum[:8])%width)
		}
		schedules[i] = Schedule{Trial: i + 1, Seed: hex.EncodeToString(sum[:]), DelayMS: delay}
	}
	return schedules, nil
}
func (p Plan) Digest() (string, error) {
	if e := p.Validate(); e != nil {
		return "", e
	}
	s, e := json.Marshal(p)
	if e != nil {
		return "", e
	}
	sum := sha256.Sum256(s)
	return hex.EncodeToString(sum[:]), nil
}
