func (s *SyncStatusSubscription) Unsubscribe() {
	s.unsubOnce.Do(func() {
		req := uninstallSyncSubscriptionRequest{s.c, make(chan interface{})}
		send := s.api.uninstallSyncSubscription
		for {
			select {
			case send <- &req:
				send = nil
			case <-s.c:
			case <-req.uninstalled:
				return
			}
		}
	})
}
