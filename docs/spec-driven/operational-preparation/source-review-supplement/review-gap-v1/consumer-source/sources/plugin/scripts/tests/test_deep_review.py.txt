class DeepReviewTests(unittest.TestCase):
 def test_actual_source_shape_and_inferred_unknown_split(self):
  b=bundle();report=review.build(b,judgments());review.validate(report,b)

 def test_canonical_hash_is_stable(self):
  b=bundle();reordered=json.loads(json.dumps(b,sort_keys=True));self.assertEqual(review.build(b,judgments()),review.build(reordered,judgments()))
 def test_missing_required_dimensions_and_all_unknown_placeholder_rejected(self):

   j=judgments();j['sections']['guards_exceptions'][field]=value
   with self.assertRaises(review.Invalid):review.build(bundle(),j)

 def test_unsupported_truncated_unknown_pack_and_bound_rejected(self):
  for change in ['truncated','base','rules','pack','bound']:
   b=bundle()

 def make_workspace(self,root):
  workspace=Path(root)/'.coding-agent/tickets/REVIEW-FIXTURE';workspace.mkdir(parents=True);review.write_review(workspace,bundle(),judgments());raw=(workspace/'deep-review.json').read_bytes();state={'ticket_id':'REVIEW-FIXTURE','ticket_type':'code_review','current_state':'PLANNING','states':{'ANALYSIS':{'status':'completed','deep_review_sha256':review.sha(raw)},'PLANNING':{'status':'pending'}}};(workspace/'state.json').write_text(json.dumps(state));return workspace,state
 def hook(self,workspace,state):
  event={'tool_name':'Write','cwd':str(workspace),'tool_input':{'file_path':str(workspace/'state.json'),'content':json.dumps(state)}}
  result=subprocess.run([sys.executable,str(ROOT/'hooks/deep-review-guard.py')],input=json.dumps(event),text=True,capture_output=True);self.assertEqual(result.returncode,0,result.stderr);return result.stdout
 def test_delivery_hash_hook_cli_and_diagrams(self):
  with tempfile.TemporaryDirectory()as root:
   workspace,state=self.make_workspace(root);review.verify_workspace(workspace);self.assertEqual(self.hook(workspace,state),'')
   state['current_state']='COMPLETED';self.assertIn('deny',self.hook(workspace,state))
   state['states']['PLANNING']={'status':'completed','review_report_sha256':review.sha((workspace/'review-report.md').read_bytes())};self.assertEqual(self.hook(workspace,state),'');review.verify_workspace(workspace,state)
   out=subprocess.run([sys.executable,str(ROOT/'scripts/deep_review.py'),'verify','--workspace',str(workspace)],capture_output=True,text=True);self.assertEqual(out.returncode,0,out.stdout)
   (workspace/'modules.mmd').unlink();self.assertIn('deny',self.hook(workspace,state))
