func must(e error){if e!=nil{panic(e)}}
func sha(b []byte)string{d:=sha256.Sum256(b);return hex.EncodeToString(d[:])}
func save(p string,x any){b,e:=json.MarshalIndent(x,"","  ");must(e);must(os.WriteFile(p,append(b,'\n'),0600))}
type ref struct{Path string `json:"path"`;First int `json:"first"`;Last int `json:"last"`;FileSHA string `json:"file_sha256"`;SpanSHA string `json:"content_sha256"`}

func sourceCheck(v string,r ref)error{
 b,_,_,e:=setup.ReadRetainedFile(v,"repo",r.Path);if e!=nil{return e};if sha(b)!=r.FileSHA{return fmt.Errorf("retained file SHA mismatch")};parts:=bytes.SplitAfter(b,[]byte{'\n'});if len(parts)>0&&len(parts[len(parts)-1])==0{parts=parts[:len(parts)-1]};if r.First<1||r.Last<r.First||r.Last>len(parts){return fmt.Errorf("invalid retained range")};if sha(bytes.Join(parts[r.First-1:r.Last],nil))!=r.SpanSHA{return fmt.Errorf("retained span SHA mismatch")};return nil
}
