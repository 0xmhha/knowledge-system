results=[]

 config=root/(name+'.yaml');config.write_text(base+setting+'\n');command=[str(bins/'cks'),'setup','--config',str(config),'--version',name,'--progress','json',*extra]
 start=time.monotonic();r=subprocess.run(command,capture_output=True,text=True,timeout=45);elapsed=time.monotonic()-start
 (e/(name+'.stdout.log')).write_text(r.stdout);(e/(name+'.stderr.log')).write_text(r.stderr)

 if code:

 results.append({'name':name,'argv':command,'exit_code':r.returncode,'elapsed_seconds':elapsed,'expected_code':code,'resources':resource,'prior_current_preserved':bool(code)})
 print(name,'PASS',round(elapsed,3),flush=True);return resource
