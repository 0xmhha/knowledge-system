import copy,json,pathlib,subprocess,tempfile
root=pathlib.Path.cwd();folder=root/'system/eval/b0-knowledge-system/refactoring-progress-guard-2026-10-06'
base=json.loads((root/'docs/spec-driven/REFACTORING-PROGRESS.json').read_text())
variants={}
a=copy.deepcopy(base);a['tasks'][3]['state']='complete';variants['false_N04_completion']=a
a=copy.deepcopy(base);a['tasks'][1]['manifest']='system/eval/b0-knowledge-system/refactoring-n02-2026-10-06/missing.json';variants['missing_committed_evidence']=a

a=copy.deepcopy(base);a['tasks'][2]['required_checks'].pop();variants['omitted_acceptance_condition']=a
report=[]
for name,ledger in variants.items():
 with tempfile.TemporaryDirectory(prefix='cks-progress-negative-') as tmp:
  path=pathlib.Path(tmp)/'progress.json';path.write_text(json.dumps(ledger))
  command=['python3','scripts/check-refactoring-progress.py','--ledger',str(path)]

  print(name,'correctly rejected',flush=True)
(folder/'negative-controls.json').write_text(json.dumps(report,indent=2)+'\n')
