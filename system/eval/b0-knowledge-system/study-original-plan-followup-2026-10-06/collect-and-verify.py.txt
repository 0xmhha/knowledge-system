sha = lambda b: hashlib.sha256(b).hexdigest()

def run(args, cwd):
    p = subprocess.run(args, cwd=cwd, text=True, capture_output=True)
    return {'argv': args, 'exit_code': p.returncode, 'stdout': p.stdout, 'stderr': p.stderr}

def githead(p):
    r = run(['git', 'rev-parse', 'HEAD'], p)
    assert r['exit_code'] == 0
    return r['stdout'].strip()

bindings = []
for name in sources:
    data = (study / name).read_bytes()
    committed = subprocess.check_output(['git', 'show', study_head + ':' + name], cwd=study)
    assert data == committed, name
    item = {'path': name, 'sha256': sha(data), 'bytes': len(data), 'matches_study_head': True}
    if Path(name).name in snapshots:
        dest = out / 'archive/study-snapshots' / name
        dest.parent.mkdir(parents=True, exist_ok=True)
        if Path(name).name == '00_background_goal.md':
            dest = Path(str(dest) + '.json')
            dest.write_text(json.dumps({'encoding': 'UTF-8', 'source_sha256': sha(data),
                'source_text': data.decode('utf-8')}, ensure_ascii=False, indent=2) + '\n')
            assert json.loads(dest.read_text())['source_text'].encode('utf-8') == data
            item['snapshot_format'] = 'JSON UTF-8 text, preserving original trailing whitespace'
        else:
            dest.write_bytes(data)
        item['snapshot'] = str(dest.relative_to(out))
    bindings.append(item)
