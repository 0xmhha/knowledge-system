package main
import("context";"encoding/json";"os";"fmt";"strings";"github.com/0xmhha/knowledge-system/internal/setup";"github.com/0xmhha/knowledge-system/internal/system/config";"github.com/0xmhha/knowledge-system/internal/system/composer/sanitize";"github.com/0xmhha/knowledge-system/internal/system/evidencev2";"github.com/0xmhha/knowledge-system/pkg/system/contract")
func main(){var in struct{Query string `json:"query"`; Refs []contract.Citation `json:"refs"`}; raw,err:=os.ReadFile(os.Args[1]);if err!=nil{panic(err)};if err=json.Unmarshal(raw,&in);err!=nil{panic(err)};spans:=[]setup.RetainedLineSpan{};for _,c:=range in.Refs{spans=append(spans,setup.RetainedLineSpan{OriginID:"repo",Path:c.File,First:c.StartLine,Last:c.EndLine})};retained,err:=setup.ReadRetainedLineSpans(os.Args[2],spans);if err!=nil{panic(err)};total:=0;for _,s:=range retained{total+=len(s.Text)};rules,err:=config.LoadSanitizeRuleset(os.Args[3]);if err!=nil{panic(err)};cleaner,err:=sanitize.New(rules);if err!=nil{panic(err)};_,err=evidencev2.Build(context.Background(),os.Args[2],in.Query,in.Refs,cleaner);message:="";if err!=nil{message=err.Error()};json.NewEncoder(os.Stdout).Encode(map[string]any{"retained_spans_verified":len(retained),"retained_bytes":total,"build_error":message});if !strings.HasPrefix(message,"budget_exceeded:"){fmt.Fprintln(os.Stderr,"unexpected diagnostic outcome");os.Exit(2)}}
